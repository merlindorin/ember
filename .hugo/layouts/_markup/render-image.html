{{- /* Images in ember/docs are relative to the Markdown file, as on GitHub. An image with a title is a window capture: it is drawn inside a window frame with that title, at a size proportional to the capture. */ -}} {{- $source := .PageInner.Params.source -}} {{- $src := .Destination -}} {{- with and $source (resources.Get (path.Join "docs" (path.Dir $source) .Destination)) -}} {{- $small := . -}} {{- $img := "" -}} {{- if eq .MediaType.SubType "svg" -}} {{- /* A diagram: vector, served as it is at the page's width. */ -}} {{- $img = printf `` .RelPermalink ($.PlainText | htmlEscape) -}} {{- else if eq .MediaType.SubType "gif" -}} {{- /* An animated capture: served as it is, since resizing would drop its frames. */ -}} {{- $img = printf `` .RelPermalink .Width .Height ($.PlainText | htmlEscape) -}} {{- else if le .Width 128 -}} {{- /* Pixel art, such as a sprite sheet: served as is and scaled up with sharp pixels. */ -}} {{- $img = printf `` .RelPermalink (mul .Width 4) (mul .Height 4) ($.PlainText | htmlEscape) -}} {{- else -}} {{- $small = .Resize (printf "%dx webp q82" (cond (lt .Width 960) .Width 960)) -}} {{- $large := .Resize (printf "%dx webp q82" .Width) -}} {{- $img = printf `` $small.RelPermalink $small.RelPermalink $large.RelPermalink $small.Width $small.Height ($.PlainText | htmlEscape) -}} {{- end -}} {{- with $.Title -}} {{ . }}{{ $img | safeHTML }} {{- else -}} {{- $img | safeHTML -}} {{- end -}} {{- else -}} {{- /* A picture task docs:shots hasn't made yet is a plain link when the build allows it (shotsOptional, the guide's tests), an error otherwise. */ -}} {{- if and $source (not site.Params.shotsOptional) -}} {{- errorf "%s: no image %q" $source .Destination -}} {{- end -}} {{- end -}}
